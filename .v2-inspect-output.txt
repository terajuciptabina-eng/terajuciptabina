  activeItems.forEach(item => roomSubtotals[item.roomId] = (roomSubtotals[item.roomId] || 0) + item.amount);
  const roomTotal = Object.values(roomSubtotals).reduce((s, v) => s + v, 0);
  renderEstimate(activeItems, roomGroups, roomSubtotals, projectPreliminaries, projectPreliminaries + roomTotal);
}
